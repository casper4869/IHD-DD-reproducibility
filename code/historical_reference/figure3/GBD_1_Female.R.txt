library(dplyr)
library(tidyr)
library(purrr)
library(vroom)

# 读入数据
IHD <- vroom::vroom("<LEGACY_PROJECT_ROOT>/Part0\\IHD_country.csv")
DD  <- vroom::vroom("<LEGACY_PROJECT_ROOT>/Part0\\DD_country.csv")

# 只保留1992-2021年 Female Age-standardized incidence
IHD_Female <- IHD %>%
  filter(age_name == 'Age-standardized',
         metric_name == 'Rate',
         measure_name == 'Incidence',
         sex_name == 'Female',
         year >= 1992, year <= 2021) %>%
  select(location_name, year, IHD_val = val)

DD_Female <- DD %>%
  filter(age_name == 'Age-standardized',
         metric_name == 'Rate',
         measure_name == 'Incidence',
         sex_name == 'Female',
         year >= 1992, year <= 2021) %>%
  select(location_name, year, DD_val = val)

# 合并数据
dat_merged <- inner_join(IHD_Female, DD_Female, by = c("location_name", "year"))

# 计算每个国家的皮尔逊相关系数与p值
corr_by_country <- dat_merged %>%
  group_by(location_name) %>%
  summarise(
    R = cor(IHD_val, DD_val, method = "pearson", use = "complete.obs"),
    p = cor.test(IHD_val, DD_val, method = "pearson")$p.value,
    IHD_sum = sum(IHD_val, na.rm=TRUE),
    DD_sum = sum(DD_val, na.rm=TRUE)
  ) %>%
  mutate(Cases_total = IHD_sum + DD_sum)

# 多重比较校正
corr_by_country <- corr_by_country %>%
  mutate(padj = p.adjust(p, method = "BH"))

# 只保留正相关且padj<0.05的国家
corr_positive <- corr_by_country %>%
  filter(R > 0, padj < 0.05) %>%
  arrange(desc(Cases_total))

# 输出结果
write.csv(corr_positive, "IHD_DD_Correlation_Female_Positive.csv", row.names = FALSE)

# 查看前10个高负担高相关国家
print(head(corr_positive, 10))
