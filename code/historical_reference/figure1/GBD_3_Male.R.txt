# 正确读取并删除第一列
dd_raw <- read.csv("grouped_countries_Male_DD.csv", header = FALSE)
dd_raw <- dd_raw[-1, ]   # 删除第一列

ihd_raw <- read.csv("grouped_countries_Male_IHD.csv", header = FALSE)
ihd_raw <- ihd_raw[-1, ]  # 删除第一列

# 添加档位编号
colnames(dd_raw) <- paste0("DD_", 1:4)
colnames(ihd_raw) <- paste0("IHD_", 1:4)

# 转为长格式 + 添加分档编号
dd_long <- dd_raw %>%
  pivot_longer(cols = everything(), names_to = "group", values_to = "location_name") %>%
  mutate(mark1 = as.integer(gsub("DD_", "", group))) %>%
  select(location_name, mark1)

ihd_long <- ihd_raw %>%
  pivot_longer(cols = everything(), names_to = "group", values_to = "location_name") %>%
  mutate(mark2 = as.integer(gsub("IHD_", "", group))) %>%
  select(location_name, mark2)

# 合并两个数据框
match_df <- full_join(dd_long, ihd_long, by = "location_name")

# 输出 CSV
write.csv(match_df, "Match_Male.csv", row.names = FALSE)
